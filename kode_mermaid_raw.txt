=======================================================
DIAGRAM 3 — Flow Process Fuzzy AHP (Weighting)
=======================================================
flowchart TD
    A(["📋 Expert Input\nPairwise Comparison Matrix\nSaaty Scale 1–9"])
    B["🔢 Fuzzification\nConvert Crisp Values\nto TFN [l, m, u]"]
    C["📊 TFN Matrix n×n\nEach cell contains\n[l, m, u]"]
    D["➕ Calculate Row Sums\nRowSum_i = Σ TFN_ij\n= [Σl, Σm, Σu]"]
    E["🔢 Calculate Total Sum\nTotalSum = Σ RowSum_i"]
    F["🔄 Calculate Inverse Total\nInvTotal = [1/u, 1/m, 1/l]"]
    G["📐 Fuzzy Synthetic Extent\nSi = RowSum_i ⊗ InvTotal"]
    H["📉 Defuzzification CoA\nDi = (li + mi + ui) / 3"]
    I["⚖️ Normalization\nWi = Di / ΣDi\n(so that ΣWi = 1)"]
    J{"Consistency Ratio\nCR = CI / RI\nCR < 0.10?"}
    K(["✅ Weights Accepted\n& Saved to DB"])
    L["❌ CR ≥ 0.10\nRevise Expert Input"]

    subgraph OUTPUT["Weight Output"]
        O1["Level 1\nGlobal Weights CP1–CP9"]
        O2["Level 2\nSub-Criteria Weights\nper CP"]
    end

    A --> B --> C --> D --> E --> F --> G --> H --> I --> J
    J -->|"Yes ✅"| K
    J -->|"No ❌"| L
    L -->|"Revise"| A
    K --> OUTPUT

    style A fill:#2C3E50,color:#fff
    style K fill:#27AE60,color:#fff
